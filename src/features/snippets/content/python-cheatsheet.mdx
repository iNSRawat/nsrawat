## Loops

Repeat code multiple times:

```python

for i in range(5):
    print(i)                                           # 0, 1, 2, 3, 4

# Range with start, stop, step
for i in range(2, 10, 2):
    print(i)                                           # 2, 4, 6, 8

# Loop through list
fruits = ["apple", "banana", "orange"]
for fruit in fruits:
    print(fruit)

# Loop with index
for index, fruit in enumerate(fruits):
    print(index, fruit)

    print(char)

# Loop through dictionary
person = {"name": "John", "age": 30}
for key, value in person.items():
    print(key, value)

count = 0
while count < 5:
    print(count)
    count += 1

# Break (exit loop)
for i in range(10):
    if i == 5:
        break
    print(i)

# Continue (skip iteration)
for i in range(5):
    if i == 2:
        continue
    print(i)                                           # 0, 1, 3, 4

    print(i)
else:
    print("Loop completed")

# Nested loops
for i in range(3):
    for j in range(3):
        print(i, j)
```

## Classes & OOP

Object-oriented programming:

```python
# Define class
class Person:
    # Constructor
    def __init__(self, name, age):
        self.name = name
        self.age = age

    # Method
    def greet(self):
        return f"Hello, my name is {self.name}"

    # String representation
    def __str__(self):
        return f"Person({self.name}, {self.age})"

# Create object
person = Person("John", 30)
print(person.name)                                     # "John"
print(person.greet())                                  # "Hello, my name is John"

# Inheritance
class Student(Person):
    def __init__(self, name, age, school):
        super().__init__(name, age)                    # Call parent constructor
        self.school = school

    def study(self):
        return f"{self.name} is studying"

student = Student("Jane", 20, "MIT")
print(student.greet())                                 # Inherited method
print(student.study())                                 # New method

# Class variables (shared across instances)
class Counter:
    count = 0                                          # Class variable

    def __init__(self):
        Counter.count += 1

# Private attributes (convention: prefix with underscore)
class BankAccount:
    def __init__(self, balance):
        self._balance = balance                        # "Private" by convention

    def get_balance(self):
        return self._balance

# Property decorator (getter/setter)
class Circle:
    def __init__(self, radius):
        self._radius = radius

    @property
    def radius(self):
        return self._radius

    @radius.setter
    def radius(self, value):
        if value < 0:
            raise ValueError("Radius cannot be negative")
        self._radius = value

circle = Circle(5)
circle.radius = 10                                     # Uses setter

# Static method (doesn't need instance)
class MathHelper:
    @staticmethod
    def add(a, b):
        return a + b

MathHelper.add(3, 5)                                   # Can call without creating instance

# Class method
class Person:
    count = 0

    @classmethod
    def get_count(cls):
        return cls.count

# Check instance type
isinstance(person, Person)                             # True
```

## Error Handling

Handle errors gracefully:

```python
# Try-except
try:
    result = 10 / 0
except ZeroDivisionError:
    print("Cannot divide by zero")

# Multiple exceptions
try:
    value = int("abc")
except ValueError:
    print("Invalid value")
except TypeError:
    print("Type error")

# Catch all exceptions
try:
    risky_operation()
except Exception as e:
    print(f"Error: {e}")

# Finally (always runs)
try:
    file = open("file.txt")
except FileNotFoundError:
    print("File not found")
finally:
    print("This always runs")

# Else (runs if no exception)
try:
    result = 10 / 2
except ZeroDivisionError:
    print("Error")
else:
    print("Success:", result)

# Raise custom exception
def check_age(age):
    if age < 0:
        raise ValueError("Age cannot be negative")
    return age

# Custom exception class
class CustomError(Exception):
    pass

raise CustomError("Something went wrong")
```
